cmake_minimum_required(VERSION 3.16) #set the minimum required version of CMake to 3.16



project(http_server VERSION 1.0.0 LANGUAGES C) #Define the project name, version, and programming language (C)

set(CMAKE_C_STANDARD 11 #Set the C standard to C11
    CMAKE_C_STANDARD_REQUIRED ON #Require the specified C standard
    CMAKE_C_EXTENSIONS OFF #Disable compiler-specific extensions
)

add_executable(http_server src/main.c) #Create an executable target named http_server from the source file main.c

target_compile_options(http_server PRIVATE -Wall -Wextra -Wpedantic) # Add compiler options to the http_server target for warnings
